cmake_minimum_required(VERSION 3.20)
project(PCNet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake" ${CMAKE_MODULE_PATH})
enable_testing()
add_executable(pcnet_p0_probe tools/p0_probe.cpp)
target_compile_features(pcnet_p0_probe PRIVATE cxx_std_17)
find_package(Qt6 REQUIRED COMPONENTS Core Core5Compat)
find_package(Isis10 REQUIRED)
target_link_libraries(pcnet_p0_probe PRIVATE Isis10::isis Qt6::Core Qt6::Core5Compat)
add_executable(p0_probe_test tests/p0_probe_test.cpp)
add_test(NAME p0_probe COMMAND p0_probe_test "${CMAKE_CURRENT_BINARY_DIR}/p0_probe.json")
add_test(NAME p0_probe_generate
  COMMAND pcnet_p0_probe --env "${CMAKE_PREFIX_PATH}" --output "${CMAKE_CURRENT_BINARY_DIR}/p0_probe.json")
set_tests_properties(p0_probe_generate PROPERTIES FIXTURES_SETUP p0_probe_report)
set_tests_properties(p0_probe PROPERTIES FIXTURES_REQUIRED p0_probe_report)
add_custom_target(run_p0_probe
  COMMAND pcnet_p0_probe --env "${CMAKE_PREFIX_PATH}" --output "${CMAKE_CURRENT_BINARY_DIR}/p0_probe.json"
  DEPENDS pcnet_p0_probe
  USES_TERMINAL)

add_library(pcnet_core STATIC
  src/core/network_view.cpp
  src/core/storage.cpp
  src/core/error_model.cpp)
target_include_directories(pcnet_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(pcnet_core PUBLIC cxx_std_17)
add_executable(network_view_test tests/core/network_view_test.cpp)
target_link_libraries(network_view_test PRIVATE pcnet_core)
add_test(NAME network_view COMMAND network_view_test)
add_executable(storage_test tests/core/storage_test.cpp)
target_link_libraries(storage_test PRIVATE pcnet_core)
add_test(NAME storage COMMAND storage_test)
add_executable(temporal_support_test tests/core/temporal_support_test.cpp)
target_link_libraries(temporal_support_test PRIVATE pcnet_core)
add_test(NAME temporal_support COMMAND temporal_support_test)
add_executable(timing_policy_test tests/core/timing_policy_test.cpp)
target_link_libraries(timing_policy_test PRIVATE pcnet_core)
add_test(NAME timing_policy COMMAND timing_policy_test)
add_executable(model_metrics_test tests/core/model_metrics_test.cpp)
target_link_libraries(model_metrics_test PRIVATE pcnet_core)
add_test(NAME model_metrics COMMAND model_metrics_test)
add_library(pcnet_metrics STATIC src/metrics/quality.cpp)
target_link_libraries(pcnet_metrics PUBLIC pcnet_core)
target_include_directories(pcnet_metrics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(quality_report_test tests/metrics/quality_report_test.cpp)
target_link_libraries(quality_report_test PRIVATE pcnet_metrics)
add_test(NAME quality_report COMMAND quality_report_test)
add_executable(report_schema_test tests/metrics/report_schema_test.cpp)
target_link_libraries(report_schema_test PRIVATE pcnet_metrics)
add_test(NAME report_schema COMMAND report_schema_test)
add_library(pcnet_isis_adapter STATIC src/adapters/isis/controlnet_adapter.cpp)
target_link_libraries(pcnet_isis_adapter PUBLIC pcnet_core Isis10::isis Qt6::Core Qt6::Core5Compat)
target_include_directories(pcnet_isis_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(isis_roundtrip_test tests/adapters/isis_roundtrip_test.cpp)
target_link_libraries(isis_roundtrip_test PRIVATE pcnet_isis_adapter Isis10::isis Qt6::Core Qt6::Core5Compat)
add_test(NAME isis_roundtrip COMMAND isis_roundtrip_test)
add_library(pcnet_thinning STATIC src/thinning/thinner.cpp)
target_link_libraries(pcnet_thinning PUBLIC pcnet_core)
target_include_directories(pcnet_thinning PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")

target_link_libraries(thinning_constraints_test PRIVATE pcnet_thinning)
add_test(NAME thinning_constraints COMMAND thinning_constraints_test)
add_executable(thinning_determinism_test tests/core/thinning_determinism_test.cpp)
target_link_libraries(thinning_determinism_test PRIVATE pcnet_thinning)
add_test(NAME thinning_determinism COMMAND thinning_determinism_test)
add_library(pcnet_filtering STATIC src/filtering/outlier_filter.cpp)
